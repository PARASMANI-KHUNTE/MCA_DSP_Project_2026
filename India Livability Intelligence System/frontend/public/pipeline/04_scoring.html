4. Factor score breakdown¶

In [5]:
factor_cols = [c for c in scored.columns if c.endswith('_score')]
scored_complete = scored_named.dropna(subset=['score'])

if not factor_cols:
    print('No factor score columns found; skipping factor breakdown plot.')
elif scored_complete.empty:
    print('No place has every weighted factor available; skipping factor breakdown plot.')
    available = [c.replace('_score', '') for c in factor_cols if scored_named[c].notna().any()]
    missing = scored_named['missing_factors'].explode().dropna().value_counts().to_dict()
    print('Available factors:', available)
    print('Missing factor counts:', missing)
else:
    top10 = scored_complete.nsmallest(10, 'rank')
    breakdown = top10[['place_name'] + factor_cols].set_index('place_name')
    breakdown.columns = [c.replace('_score','').replace('_',' ').title() for c in breakdown.columns]
    breakdown.plot(kind='bar', figsize=(12,5), colormap='tab10', edgecolor='white')
    plt.title('Factor Score Breakdown - Top 10 Places')
    plt.ylabel('Factor Score (0-100)')
    plt.xticks(rotation=30, ha='right')
    plt.legend(bbox_to_anchor=(1.01,1), loc='upper left', fontsize=9)
    plt.tight_layout()
    plt.savefig(f'{FIGURES}/04_factor_breakdown.png', dpi=150, bbox_inches='tight')
    plt.show()

presets = {'Balanced': {'air_quality':0.30,'safety':0.25,'health':0.25,'economy':0.20}, 'Clean Air': {'air_quality':0.55,'safety':0.15,'health':0.20,'economy':0.10}, 'Family': {'air_quality':0.20,'safety':0.30,'health':0.30,'economy':0.20}, 'Career': {'air_quality':0.15,'safety':0.15,'health':0.10,'economy':0.60}, 'Retirement': {'air_quality':0.30,'safety':0.15,'health':0.45,'economy':0.10}}
rank_rows = []
for preset_name, w in presets.items():
    res = compute_scores(wide, config, w, year=None)
    res['preset'] = preset_name
    rank_rows.append(res[['place_id','rank','preset']])
rank_df = pd.concat(rank_rows, ignore_index=True)
rank_named = rank_df.merge(places[['place_id','place_name','place_type']], on='place_id', how='left')
# Delhi exists as both a state and a city, so disambiguate the label before pivoting.
rank_named['label'] = rank_named['place_name'] + rank_named['place_type'].map(
    {'state': ' (state)', 'city': ' (city)'}
).fillna('')
rank_pivot = rank_named.pivot_table(
    index='label', columns='preset', values='rank', aggfunc='min'
)
if rank_pivot.dropna().empty:
    print('No ranked places available for the current factors; skipping preset comparison.')
else:
    print(rank_pivot.dropna().nsmallest(15, 'Balanced'))
