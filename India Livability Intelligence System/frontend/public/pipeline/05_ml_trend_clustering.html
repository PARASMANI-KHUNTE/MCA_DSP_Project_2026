# Independent target: inherited health score; predictors are city air, economy, and region.
from statsmodels.stats.diagnostic import het_breuschpagan
from statsmodels.stats.outliers_influence import variance_inflation_factor
from scipy import stats

latest_yr = int(scores_long['year'].max())
latest = scores_long[scores_long['year'] == latest_yr].merge(
    places[['place_id', 'region', 'place_type']], on='place_id', how='left'
)
latest = latest[latest['place_type'] == 'city'].copy()
fpreds = [c for c in ['air_quality_score', 'economy_score'] if c in latest.columns]
if 'health_score' in latest.columns and len(fpreds) == 2:
    y = pd.to_numeric(latest['health_score'], errors='coerce')
    X = latest[fpreds].apply(pd.to_numeric, errors='coerce')
    rdum = pd.get_dummies(latest['region'], prefix='region', drop_first=True, dtype=float)
    X = pd.concat([X, rdum], axis=1)
    mask = y.notna() & X.notna().all(axis=1)
    X = X.loc[mask].astype(float)
    y = y.loc[mask].astype(float)
    if len(y) >= max(8, X.shape[1] + 3):
        Xc = sm.add_constant(X, has_constant='add')
        mlr = sm.OLS(y, Xc).fit()
        print(mlr.summary())
        print(f'R-squared={mlr.rsquared:.3f}; adjusted R-squared={mlr.rsquared_adj:.3f}')
        vif = pd.DataFrame({
            'feature': Xc.columns,
            'VIF': [variance_inflation_factor(Xc.to_numpy(), i) for i in range(Xc.shape[1])],
        })
        print(vif)
        bp_lm, bp_p, _, _ = het_breuschpagan(mlr.resid, Xc)
        print(f'Breusch-Pagan LM={bp_lm:.3f}, p={bp_p:.4f}')
        fig, axes = plt.subplots(1, 2, figsize=(10, 4))
        axes[0].scatter(mlr.fittedvalues, mlr.resid)
        axes[0].axhline(0, color='black', linewidth=1)
        axes[0].set(xlabel='Fitted health score', ylabel='Residual', title='Residuals vs fitted')
        stats.probplot(mlr.resid, dist='norm', plot=axes[1])
        axes[1].set_title('Residual Q-Q plot')
        plt.tight_layout()
        plt.show()
    else:
        print(f'Insufficient complete city rows for MLR diagnostics (n={len(y)}).')
else:
    print('Required health, city-air, or economy score columns are unavailable.')

5. Train/Validate/Test split (M6.3)¶

Time-based split (never random). Air: train=2015-2018, val=2019, test=2023-2024.

In [7]:
def _design(years):
    years = np.asarray(years, dtype=float).reshape(-1, 1)
    return sm.add_constant(years, has_constant="add")


metrics_all = {}
for pid in fslong[fslong['factor']=='air_quality']['place_id'].unique()[:8]:
    sub = fslong[(fslong['place_id']==pid) & (fslong['factor']=='air_quality')].dropna(subset=['score']).sort_values('year')
    if len(sub)<5: continue
    train = sub[sub['year']<=2018]; val = sub[sub['year']==2019]; test = sub[sub['year']>=2023]
    if len(train)<3: continue
    m = sm.OLS(train['score'].values, _design(train['year'].values)).fit()
    res = {'place_id': pid}
    if not val.empty:
        yv = m.predict(_design(val['year'].values))
        res['val_rmse'] = float(np.sqrt(mean_squared_error(val['score'].values, yv)))
    if not test.empty:
        yt = m.predict(_design(test['year'].values))
        res['test_rmse'] = float(np.sqrt(mean_squared_error(test['score'].values, yt)))
        nb_pred = np.full(len(test), float(train['score'].iloc[-1]))
        res['baseline_rmse'] = float(np.sqrt(mean_squared_error(test['score'].values, nb_pred)))
    metrics_all[pid] = res
metrics_df_m = pd.DataFrame(metrics_all).T
print('Validation metrics:')
print(metrics_df_m.round(3))

Summary¶

This notebook:

  • Fitted OLS linear trends per (place, factor) with actual year values to handle gaps correctly.
  • Classified trends as Improving / Declining / Stable or uncertain (slope p-value, alpha=0.05).
  • Generated 10-year projections with 95% prediction intervals, clipped to [0, 100].
  • Used time-based split (never random): train 2015-2018, val 2019, test 2023-2024.
  • Compared against naive baseline (last observed value repeated).
  • Clustered places into livability tiers with K-Means.

Wide intervals are expected and honest with only 5-10 data points per series.
